Here is the complete Google Apps Script to replace your heavy formulas.

Because Google Apps Script cannot reliably handle 1,250 simultaneous `GOOGLEFINANCE` calls without timing out, this script uses the **Yahoo Finance API** to fetch the historical data in the background. It calculates the 50, 100, and 200 DMAs, perfectly replicates your advanced Cumulative Average Rule (CAR) logic, and then pastes the final results into your sheet as **lightning-fast static values**.

### How to Install and Run the Script

1.  In your Google Sheet, click on **Extensions \> Apps Script** in the top menu.
2.  Delete any code in the editor and paste the code provided below.
3.  Click the **Save** icon (floppy disk) at the top.
4.  Click the **Run** button at the top to execute the script.
    *(Note: The first time you run it, Google will ask you to authorize the script. Click "Review permissions", choose your Google account, click "Advanced", and then "Go to project".)*

### The Apps Script Code

``` javascript
/**
 * Updates the "Top 250 Stocks" sheet with static values for DMAs and CAR Rating.
 * This completely eliminates #N/A errors and sheet lag.
 */
function updateStockData() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Top 250 Stocks");
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;
  
  // Get all tickers and existing data from Columns A to H
  const dataRange = sheet.getRange(2, 1, lastRow - 1, 8);
  const data = dataRange.getValues();
  const results = [];
  
  for (let i = 0; i < data.length; i++) {
    const nseCode = data[i][0];
    
    // Skip empty rows
    if (!nseCode || nseCode === "") {
      results.push([data[i][1], data[i][2], data[i][3], data[i][4], data[i][5], data[i][6], data[i][7]]);
      continue;
    }
    
    // Format ticker for Yahoo Finance API (e.g., "NSE:RELIANCE" -> "RELIANCE.NS")
    const symbol = nseCode.replace("NSE:", "").trim() + ".NS";
    
    try {
      // Fetch 1 year of daily historical data
      const url = `https://query1.finance.yahoo.com/v8/finance/chart/${symbol}?interval=1d&range=1y`;
      const response = UrlFetchApp.fetch(url, {muteHttpExceptions: true});
      
      if (response.getResponseCode() !== 200) {
        throw new Error("API Error");
      }
      
      const json = JSON.parse(response.getContentText());
      const result = json.chart.result[0];
      const closePrices = result.indicators.quote[0].close;
      const highPrices = result.indicators.quote[0].high;
      
      // Filter out null values (weekends/holidays)
      const validCloses = closePrices.filter(p => p !== null);
      const validHighs = highPrices.filter(p => p !== null);
      
      // Check if the stock has enough history
      if (validCloses.length < 200) {
        results.push([validCloses[validCloses.length - 1] || "N/A", "N/A", "N/A", "N/A", "Unconfirmed", "N/A", "Short History"]);
        continue;
      }
      
      const cmp = validCloses[validCloses.length - 1];
      
      // Calculate 50, 100, 200 DMAs
      const dma50 = average(validCloses.slice(-50));
      const dma100 = average(validCloses.slice(-100));
      const dma200 = average(validCloses.slice(-200));
      
      // Output Trend (Bull/Bear Run)
      let output = "Unconfirmed";
      if (cmp > dma50 && dma50 > dma100 && dma100 > dma200) {
        output = "In Bull Run";
      } else if (cmp < dma50 && dma50 < dma100 && dma100 < dma200) {
        output = "In Bear Run";
      }
      
      // Difference from 200 DMA
      const diff200 = ((cmp - dma200) / dma200) * 100;
      
      // --- CAR Rating Logic (Replicated from your LET formula) ---
      // 1. Find 52-week high index
      let maxHigh = -1;
      let maxHighIndex = -1;
      for (let j = 0; j < validHighs.length; j++) {
         if (validHighs[j] > maxHigh) {
            maxHigh = validHighs[j];
            maxHighIndex = j;
         }
      }
      
      // 2. Get closes since the high date
      const closesSinceHigh = validCloses.slice(maxHighIndex);
      let carRating = "Avoid/Hold";
      
      if (closesSinceHigh.length < 10) {
         carRating = "Short History";
      } else {
         // 3. Calculate cumulative averages
         const cumAvgs = [];
         let sum = 0;
         for (let j = 0; j < closesSinceHigh.length; j++) {
            sum += closesSinceHigh[j];
            cumAvgs.push(sum / (j + 1));
         }
         
         // 4. Compare latest cumulative average to the previous 9
         const last10 = cumAvgs.slice(-10);
         const latestCumAvg = last10[last10.length - 1];
         
         let checkCount = 1; // Starts at 1 to match the dummy '0' in original formula
         for (let k = 0; k < last10.length - 1; k++) {
            if (latestCumAvg > last10[k]) {
               checkCount++;
            }
         }
         
         if (checkCount === 10) {
            carRating = "Buy/Average Out";
         }
      }
      
      // Add row to results
      results.push([cmp, dma50, dma100, dma200, output, diff200, carRating]);
      
    } catch(e) {
      // If ticker fails or is missing, flag it but keep existing data layout
      results.push([data[i][1], data[i][2], data[i][3], data[i][4], data[i][5], data[i][6], "TICKER NOT FOUND"]);
    }
    
    // Brief pause to respect API rate limits (150 milliseconds)
    Utilities.sleep(150);
  }
  
  // Write all calculated data back to the sheet at once (Columns B through H)
  sheet.getRange(2, 2, results.length, 7).setValues(results);
}

// Helper function to calculate averages
function average(arr) {
  const sum = arr.reduce((a, b) => a + b, 0);
  return sum / arr.length;
}

```

### Next Steps: Automating It

Once you confirm it works perfectly, you can delete the heavy formulas in your sheet (the script will overwrite them with static values anyway).

To make this run automatically every day:

1.  In the Apps Script editor, click the **Clock icon** (Triggers) on the left sidebar.
2.  Click **Add Trigger** in the bottom right.
3.  Set it to run `updateStockData`, choose **Time-driven**, select **Day timer**, and pick a time (e.g., `5pm to 6pm` after the market closes).
4.  Click **Save**. Your sheet will now update itself automatically every day with zero lag\!
